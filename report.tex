\section{Giới thiệu bài toán}
% ════════════════════════════════════════════════════════════════════════════

\subsection{Quy hoạch tuyến tính}

Quy hoạch tuyến tính (\emph{linear programming}, LP) là bài toán tối ưu một
hàm mục tiêu tuyến tính trên một tập ràng buộc tuyến tính. Tập nghiệm của hệ
ràng buộc là một \textbf{đa diện lồi} (\emph{polyhedron}); lý thuyết về đa diện
là nền tảng của toàn bộ quy hoạch tuyến tính \cite{bertsimas1997}.

\begin{definition}[Bài toán LP dạng ma trận]
Bài toán LP được viết gọn dưới dạng ma trận
\begin{equation}
    \text{tối ưu}\quad z = \vc\T\vx
    \qquad\text{với}\qquad \mA\vx \le \vb,
    \label{eq:lp}
\end{equation}
trong đó $\vx \in \R^{n}$ là vector biến quyết định, $\vc \in \R^{n}$ là hệ số
hàm mục tiêu, còn $\mA \in \R^{m\times n}$, $\vb \in \R^{m}$ mô tả $m$ ràng
buộc. Điều kiện không âm $x_i \ge 0$ (nếu có) cũng chỉ là ràng buộc dạng
$-x_i \le 0$.
\end{definition}

Cách viết \eqref{eq:lp} theo \emph{dạng ma trận} là điểm khởi đầu cho mọi xử
lý tính toán: toàn bộ dữ liệu đầu vào quy về một bảng số $[\mA \mid \vb]$, và
mỗi phép biến đổi trên hệ ràng buộc trở thành một phép biến đổi trên các hàng
của bảng đó \cite{vanderbei2014}.

\subsection{Phép khử Fourier-Motzkin}

Phép khử Fourier-Motzkin là một thủ tục \emph{loại biến} cho hệ bất đẳng
thức tuyến tính. Về mặt hình học, loại một biến tương ứng với việc \textbf{chiếu}
đa diện từ không gian $n$ chiều xuống không gian $n-1$ chiều
\cite{bertsimas1997, schrijver1986}. Lặp lại phép chiếu cho từng biến, ta thu
được hình chiếu của đa diện xuống không gian con bất kỳ - từ đó đọc ra được
khoảng giá trị khả thi của hàm mục tiêu.

\subsection{Mục tiêu của bài tập lớn}

\begin{itemize}[leftmargin=1.4cm]
    \item Cài đặt thuật toán khử Fourier-Motzkin để giải bài toán LP tổng quát
          $n$ biến, làm việc hoàn toàn trên \textbf{trường số hữu tỉ} $\Q$ để
          chính xác tuyệt đối.
    \item Bổ sung \textbf{phương pháp hình học} cho bài toán hai biến nhằm trực
          quan hóa và đối chiếu kết quả.
    \item Khi phát hiện vô nghiệm, xuất \textbf{chứng chỉ Farkas} - tổ hợp không
          âm các ràng buộc gốc tạo ra mâu thuẫn.
    \item Minh họa một ứng dụng trong khoa học máy tính: dùng khử
          Fourier--Motzkin làm kiểm thử khả thi cho \textbf{phân tích phụ thuộc
          vòng lặp} \cite{allen2002}.
    \item Xây dựng ứng dụng web giải trình từng bước, kèm bộ kiểm thử tự động.
\end{itemize}

\subsection{Phạm vi}

Phép khử Fourier--Motzkin áp dụng cho mọi $n \ge 1$. Phương pháp hình học chỉ
áp dụng cho $n = 2$ vì cần biểu diễn miền nghiệm trên mặt phẳng. Ràng buộc cho
phép cả ba dạng $\le$, $\ge$, $=$; hệ số phải quy được về số hữu tỉ (chấp nhận
phân số và các hằng ký hiệu hữu tỉ như $\sin(\pi/6) = 1/2$).

% ════════════════════════════════════════════════════════════════════════════
\section{Thuật toán sử dụng}
% ════════════════════════════════════════════════════════════════════════════

\subsection{Chuẩn hóa về dạng \texorpdfstring{$\le$}{<=}}

Mọi ràng buộc được đưa về một dạng duy nhất $\mathbf{a}\T\vx \le b$ để xử lý
đồng nhất \cite{vanderbei2014}:
\begin{itemize}
    \item ràng buộc $\ge$ được nhân hai vế với $-1$;
    \item ràng buộc $=$ được tách thành hai ràng buộc $\le$ và $\ge$ (rồi cũng
          đưa về $\le$).
\end{itemize}
Sau bước này, hệ ràng buộc đúng là ma trận $[\mA \mid \vb]$ với toán tử $\le$
ở mọi hàng.

\subsection{Khử một biến: phép chiếu đa diện}

Xét hệ các bất đẳng thức $\le$ và một biến $x_k$ cần khử. Phân loại mỗi ràng
buộc theo dấu hệ số $a_{k}$ của $x_k$ trong ràng buộc đó:
\begin{itemize}
    \item $a_{k} > 0$: chia hai vế cho $a_{k}$ được một \textbf{chặn trên}
          $x_k \le U_i(\vx_{-k})$ (nhóm $U$);
    \item $a_{k} < 0$: chia hai vế cho $a_{k}$ (đổi chiều) được một
          \textbf{chặn dưới} $x_k \ge L_l(\vx_{-k})$ (nhóm $L$);
    \item $a_{k} = 0$: ràng buộc không chứa $x_k$, giữ nguyên (nhóm $N$).
\end{itemize}
Với phần biến còn lại $\vx_{-k}$ cố định, tồn tại giá trị $x_k$ thỏa mãn khi và
chỉ khi \emph{mọi chặn dưới không vượt quá mọi chặn trên}:
\begin{equation}
    \max_l L_l(\vx_{-k}) \;\le\; \min_i U_i(\vx_{-k}).
\end{equation}
Điều này tương đương với việc ghép \emph{mọi cặp} $(L_l, U_i)$:
\begin{equation}
    L_l(\vx_{-k}) \le U_i(\vx_{-k}),
    \qquad \forall\, l, i.
    \label{eq:pairing}
\end{equation}
Hệ mới gồm các ràng buộc nhóm $N$ cộng với các cặp \eqref{eq:pairing}; nó
\textbf{không còn $x_k$} và có tập nghiệm đúng bằng hình chiếu của đa diện cũ
lên các biến còn lại \cite{bertsimas1997, schrijver1986}. Lặp lại cho
$x_1, \ldots, x_n$.

\begin{infobox}[Cài đặt phép ghép cặp không cần chia]
Thay vì chia cho $a_k$ rồi cộng (dễ phát sinh phân số trung gian), cài đặt thực
tế nhân chéo: với một ràng buộc chặn dưới $r_L$ (hệ số $a_{Lk} < 0$) và một
ràng buộc chặn trên $r_U$ (hệ số $a_{Uk} > 0$), đặt
\[
    \alpha = a_{Uk} > 0, \qquad \beta = -a_{Lk} > 0,
\]
rồi tạo ràng buộc mới $\alpha\, r_L + \beta\, r_U$. Hệ số của $x_k$ trong tổ
hợp này bằng $\alpha\,a_{Lk} + \beta\,a_{Uk} = 0$, đúng như mong muốn. Vì
$\alpha, \beta > 0$, đây là một \emph{tổ hợp không âm} của hai bất đẳng thức.
\end{infobox}

\subsection{Đưa hàm mục tiêu vào hệ ràng buộc}

Để dùng phép khử cho bài toán \emph{tối ưu} (chứ không chỉ kiểm tra khả thi),
ta thêm một biến phụ $z$ và biến bài toán thành tìm cận của $z$:
\begin{itemize}
    \item Với $\max z = \vc\T\vx$: thêm ràng buộc $z - \vc\T\vx \le 0$
          (tức $z \le \vc\T\vx$).
    \item Với $\min z = \vc\T\vx$: dùng đẳng thức $\min \vc\T\vx = -\max(-\vc\T\vx)$,
          thêm ràng buộc $z + \vc\T\vx \le 0$ (tức $z \le -\vc\T\vx$); giải được
          $z_{\max}$ rồi lấy $z^\ast = -z_{\max}$.
\end{itemize}
Khử lần lượt $x_1, \ldots, x_n$ \emph{nhưng giữ lại $z$}. Sau khi khử hết, hệ
chỉ còn các ràng buộc theo $z$; giá trị tối ưu là cận trên nhỏ nhất:
\begin{equation}
    z_{\max} = \min\bigl\{\, b/a \;:\; a z \le b,\ a > 0 \,\bigr\}.
    \label{eq:zmax}
\end{equation}

\begin{infobox}[Đọc nghiệm từ hệ rút gọn]
Khi đọc cận của $z$ theo \eqref{eq:zmax}, chương trình \emph{chỉ xét những ràng
buộc mà mọi hệ số $x_j$ đều bằng $0$} (đã chiếu hoàn toàn về biến $z$). Ba khả
năng:
\begin{itemize}
    \item Xuất hiện ràng buộc hằng mâu thuẫn $0 \le b$ với $b < 0$:
          bài toán \textbf{vô nghiệm}.
    \item Không có cận trên nào cho $z$: bài toán \textbf{không bị chặn}.
    \item Ngược lại: $z_{\max}$ là cận trên nhỏ nhất.
\end{itemize}
\end{infobox}

\subsection{Khôi phục nghiệm bằng thế ngược}

Sau khi có $z_{\max}$, ta cần một điểm $\vx^\ast$ đạt giá trị đó. Chương trình
lưu \emph{hệ ràng buộc ngay trước mỗi bước khử}. Đi ngược từ $x_n$ về
$x_1$: tại biến $x_k$, thế $z = z_{\max}$ và các $x_j$ ($j > k$) đã biết vào hệ
trước khi khử $x_k$, thu được khoảng $[\text{lo}, \text{hi}]$ cho $x_k$, rồi
chọn cận dưới (ưu tiên điểm tại biên - một đỉnh của đa diện). Cuối cùng,
nghiệm tìm được luôn được thế lại vào toàn bộ hệ ràng buộc gốc để xác nhận.

\subsection{Bổ đề Farkas cho trường hợp vô nghiệm}
\label{sec:farkas}

\begin{lemma}[Farkas, dạng bất đẳng thức]
Hệ $\mA\vx \le \vb$ vô nghiệm khi và chỉ khi tồn tại vector nhân tử
$\vy \ge \mathbf{0}$ sao cho
\[
    \vy\T\mA = \mathbf{0}\quad\text{và}\quad \vy\T\vb < 0.
\]
\end{lemma}

Mỗi ràng buộc mới sinh trong quá trình khử là một \emph{tổ hợp không âm} của
các ràng buộc trước đó. Do đó, nếu bước cuối tạo ra
một bất đẳng thức mâu thuẫn $0 \le b$ với $b < 0$, thì truy ngược các hệ số tổ
hợp về hệ gốc cho đúng vector $\vy \ge \mathbf{0}$ thỏa Bổ đề Farkas. Cài
đặt lưu trong từng hàng: một ánh xạ
$\{\text{chỉ số ràng buộc gốc} \mapsto \text{nhân tử} \ge 0\}$, được cộng dồn
theo đúng trọng số $\alpha, \beta$ ở mỗi lần ghép cặp.

\subsection{Giảm bùng nổ tổ hợp bằng lọc dư thừa}
\label{sec:prune}

Điểm yếu  của Fourier--Motzkin: khi khử $x_k$, số ràng buộc mới sinh là
$|U|\cdot|L|$, có thể tới $\lfloor m^2/4\rfloor$ với $m$ là số ràng buộc hiện
tại. Sau $n$ bước, số ràng buộc có thể tăng theo hàm \emph{mũ kép}
\cite{vanderbei2014, schrijver1986}:
\begin{equation}
    m_n = O\!\left(m^{\,2^{n}}\right).
    \label{eq:blowup}
\end{equation}
Phần lớn các ràng buộc sinh ra là \textbf{dư thừa}. Chương trình áp dụng một
phép lọc \emph{an toàn} (không đổi tập nghiệm) và \emph{rẻ} sau mỗi bước khử:
\begin{itemize}
    \item bỏ ràng buộc luôn đúng $0 \le b$ với $b \ge 0$;
    \item với nhóm ràng buộc có vế trái là \emph{bội dương} của nhau (cùng một
          hướng nửa không gian), chỉ giữ ràng buộc \emph{chặt nhất} - vế phải
          nhỏ nhất sau khi quy về cùng tỉ lệ; các ràng buộc còn lại bị nó trội
          nên là dư thừa.
\end{itemize}
Phép lọc không loại được dư thừa "ẩn" (việc đó tự nó là một bài toán LP), nhưng
xử lý gọn các dư thừa hiển nhiên về cú pháp. Mục \ref{sec:growth} cho số liệu
định lượng về hiệu quả của bước này.

\subsection{Mã giả}

\begin{lstlisting}[language=Python, caption={Khử một biến (bám sát cài đặt fmlp/eliminate.py)}]
def eliminate_variable(rows, k):
    lower, upper, none = [], [], []      # chan duoi (a<0), chan tren (a>0), khong chua x_k
    for r in rows:
        a = r.coeff(k)
        (upper if a > 0 else lower if a < 0 else none).append(r)

    out = [r.copy() for r in none]
    for lo in lower:                     # ghep moi cap (chan duoi, chan tren)
        for up in upper:
            out.append(combine(lo, up, k))   # alpha*lo + beta*up, triet tieu x_k
    return out

def combine(low, up, k):
    alpha = up.coeff(k)                  # > 0
    beta  = -low.coeff(k)                # > 0
    coeffs = [alpha*low.coeff(j) + beta*up.coeff(j) for j in range(width)]
    coeffs[k] = 0
    rhs = alpha*low.rhs + beta*up.rhs
    cert = combine_certificates(low, up, alpha, beta)   # boi so Farkas cong don
    return Row(coeffs, rhs, cert)
\end{lstlisting}

\subsection{Phương pháp hình học (cho \texorpdfstring{$n = 2$}{n=2})}

Cơ sở là định lý điểm cực biên của lý thuyết đa diện \cite{bertsimas1997}.

\begin{theorem}[Điểm cực biên]
Nếu bài toán LP \eqref{eq:lp} có nghiệm tối ưu hữu hạn thì tồn tại một đỉnh
(điểm cực biên) của đa diện khả thi đạt tối ưu.
\end{theorem}

Với $n = 2$, đa diện khả thi là một đa giác lồi trong mặt phẳng. Thuật toán:
\begin{enumerate}
    \item coi mỗi ràng buộc là một đường thẳng (lấy dấu $=$);
    \item giải mọi cặp đường để tìm giao điểm; giữ lại giao điểm thỏa mọi ràng
          buộc - đó là các đỉnh;
    \item tính $z = \vc\T\vx$ tại từng đỉnh, chọn đỉnh tối ưu;
    \item trực quan bằng \emph{đường mức} $z = \text{const}$: trượt đường mức
          theo gradient $\nabla z = (c_1, c_2)$ (hướng tăng với $\max$, giảm với
          $\min$); điểm chạm cuối cùng giữa đường mức và miền khả thi là nghiệm
          tối ưu.
\end{enumerate}

\begin{infobox}[Giới hạn của phương pháp hình học]
Phương pháp hình học chỉ liệt kê các \emph{đỉnh hữu hạn}, nên không phát hiện
được trường hợp \textbf{không bị chặn}: khi đó nó chỉ trả về đỉnh tốt nhất trong
số tìm được. Trường hợp này nên dùng phương pháp đại số (phát hiện đúng tính
không bị chặn qua bước \eqref{eq:zmax}).
\end{infobox}

\section{Ví dụ minh họa thuật toán}
% ════════════════════════════════════════════════════════════════════════════

\begin{example}[Bài toán hai biến]
\label{ex:tc1}
Giải $\max\ z = 3x_1 + 2x_2$ với
$x_1 + x_2 \le 4$, $2x_1 + x_2 \le 6$, $x_1 \ge 0$, $x_2 \ge 0$.
\end{example}

\textbf{Bước 0 - Chuẩn hóa và gắn biến $z$.} Hệ ban đầu (đúng như dữ liệu
chương trình in ra ở mục \texttt{steps}, bước 0):
\begin{align*}
    x_1 + x_2 &\le 4, &
    2x_1 + x_2 &\le 6, \\
    -x_1 &\le 0, &
    -x_2 &\le 0, \\
    -3x_1 - 2x_2 + z &\le 0. & &
\end{align*}

\textbf{Bước 1 - Khử $x_1$.} Nhóm theo dấu hệ số $x_1$: chặn trên
$\{x_1+x_2\le4,\ 2x_1+x_2\le6\}$, chặn dưới $\{-x_1\le0\}$, ràng buộc còn lại
chứa $z$. Ghép cặp rồi lọc dư thừa, hệ trở thành (theo $x_2, z$):
\[
    -x_2 \le 0,\quad x_2 \le 4,\quad x_2 + z \le 12,\quad -x_2 + 2z \le 18.
\]
(Ràng buộc $x_2 \le 6$ sinh ra ở bước này bị $x_2 \le 4$ trội nên đã bị lọc bỏ
theo Mục \ref{sec:prune}.)

\textbf{Bước 2 - Khử $x_2$.} Sau khi ghép cặp tiếp và lọc dư thừa, ràng buộc
duy nhất còn lại \emph{chỉ chứa $z$} là $3z \le 30$. Theo \eqref{eq:zmax}:
\[
    z_{\max} = \tfrac{30}{3} = 10.
\]

\textbf{Thế ngược.} Cố định $z = 10$. Dùng hệ trước khi khử $x_2$ (hệ
ở Bước 1): từ $x_2 + z \le 12$ suy $x_2 \le 2$, từ $-x_2 + 2z \le 18$ suy
$x_2 \ge 2$, vậy $x_2 = 2$. Tiếp tục với hệ trước khi khử $x_1$ và
$x_2 = 2,\ z = 10$: thu được $x_1 = 2$.

\begin{resultbox}[Kết quả]
$x_1^\ast = 2,\quad x_2^\ast = 2,\quad z^\ast = 10.$
Nghiệm thỏa mọi ràng buộc gốc. Phương pháp hình học cho cùng kết quả: đỉnh tối
ưu là $A(2, 2)$ với $z = 10$.
\end{resultbox}

\begin{example}[Bài toán vô nghiệm và chứng chỉ Farkas]
Giải $\max\ z = x_1$ với $x_1 \le -1$ và $x_1 \ge 0$. Chuẩn hóa: $x_1 \le -1$
(ràng buộc \#1) và $-x_1 \le 0$ (ràng buộc \#2). Khử $x_1$: ghép cặp chặn dưới
$-x_1\le0$ với chặn trên $x_1\le-1$, nhân tử đều bằng $1$, được
\[
    1\cdot(x_1 \le -1) + 1\cdot(-x_1 \le 0)\ \Longrightarrow\ 0 \le -1,
\]
một mâu thuẫn. Chương trình kết luận \textbf{vô nghiệm} và xuất chứng chỉ
Farkas $\vy = (1, 1)$: thật vậy $\vy\T\mA = (1)(1) + (1)(-1) = 0$ và
$\vy\T\vb = (1)(-1) + (1)(0) = -1 < 0$, đúng Bổ đề Farkas.
\end{example}

\section{Ứng dụng trong khoa học máy tính: kiểm tra phụ thuộc vòng lặp}
% ════════════════════════════════════════════════════════════════════════════

Phép khử Fourier-Motzkin không chỉ để giải toán; nó là công cụ chuẩn trong các
trình biên dịch tối ưu hóa để \textbf{phân tích phụ thuộc dữ liệu vòng lặp}
\cite{allen2002}. Hiểu đúng phụ thuộc là điều kiện để trình biên dịch quyết
định có được phép \emph{hoán đổi vòng lặp} hay \emph{song song hóa} hay không.

\subsection{Mô hình tổng quát}

Xét $d$ vòng lặp lồng nhau với vector lặp $\mathbf{i} = (i_1,\ldots,i_d)$,
mỗi $i_k$ chạy trong biên $[L_k, U_k]$, và một mảng $m$ chiều được truy cập
bằng hàm chỉ số \emph{affine}:

\begin{lstlisting}[language=Python,
    caption={Mô hình vòng lặp lồng nhau tổng quát},
    escapechar=!]
for !$i_1$! in range(!$L_1$!, !$U_1$!+1):
  for !$i_2$! in range(!$L_2$!, !$U_2$!+1):
    ...
      for !$i_d$! in range(!$L_d$!, !$U_d$!+1):
          A[!$f_1$!(i), ..., !$f_m$!(i)] = ...   # GHI
          ... A[!$g_1$!(i), ..., !$g_m$!(i)] ... # DOC
\end{lstlisting}

Hàm chỉ số affine chiều $k$:
\[
    f_k(\mathbf{i}) = \sum_{j=1}^{d} a_{k,j}\,i_j + c_k^{(w)}, \qquad
    g_k(\mathbf{i}) = \sum_{j=1}^{d} b_{k,j}\,i_j + c_k^{(r)}.
\]

Tồn tại phụ thuộc khi có hai vector lặp $\mathbf{i}_w$ (ghi) và $\mathbf{i}_r$
(đọc), cả hai trong không gian lặp, sao cho chúng trỏ vào \emph{cùng phần tử
mảng} trên mọi chiều:
\begin{equation}
    f_k(\mathbf{i}_w) = g_k(\mathbf{i}_r) \quad \forall k = 1,\ldots,m,
    \qquad L_j \le i_{w,j},\, i_{r,j} \le U_j \quad \forall j.
    \label{eq:dep}
\end{equation}
Hệ \eqref{eq:dep} có $2d$ biến và $m + 4d$ ràng buộc. Câu hỏi "có phụ thuộc
không?" trở thành "hệ \eqref{eq:dep} có khả thi không?" - và khử
Fourier-Motzkin trả lời được.

\subsection{Quy trình kiểm tra hai tầng}

Thay vì chạy FM ngay, cài đặt dùng hai tầng theo thứ tự từ rẻ đến đắt:

\textbf{Tầng 1 — GCD test} (điều kiện cần, chính xác trên số nguyên). Với mỗi
chiều $k$, phương trình $\sum a_{k,j} i_{w,j} - \sum b_{k,j} i_{r,j} =
c_k^{(r)} - c_k^{(w)}$ có nghiệm nguyên khi và chỉ khi ước chung lớn nhất của
tất cả hệ số chia hết vế phải (định lý Bézout). Nếu bất kỳ chiều nào không thỏa
điều kiện này thì chắc chắn không có phụ thuộc, dừng ngay mà không cần chạy FM.

\textbf{Tầng 2 — FM trên số thực} (điều kiện cần, bảo thủ). Nếu qua
GCD test, chạy khử Fourier-Motzkin để kiểm tra tính khả thi. Kết luận:
\begin{itemize}
    \item Hệ thực \textbf{vô nghiệm} $\Rightarrow$ hệ nguyên cũng vô nghiệm
          $\Rightarrow$ \textbf{không có phụ thuộc}: trình biên dịch được tự do
          song song hóa.
    \item Hệ thực \textbf{khả thi} $\Rightarrow$ \textbf{có thể có phụ thuộc}:
          trình biên dịch phải giữ thứ tự.
\end{itemize}

\begin{example}[Các tình huống minh họa]
Với vòng lặp $i \in [0, 10]$:
\begin{itemize}
    \item \textbf{Ghi $A[i]$, đọc $A[i-1]$}: hệ $i_w = i_r - 1$ khả thi
          ($i_w{=}0, i_r{=}1$) $\Rightarrow$ có phụ thuộc.
    \item \textbf{Ghi $A[i]$, đọc $A[i+11]$}: cần $i_w = i_r + 11$ với cả hai
          trong $[0,10]$ — vô nghiệm $\Rightarrow$ không phụ thuộc.
    \item \textbf{Ghi $A[2i]$, đọc $A[2i+1]$}: GCD$(2,2)=2$ không chia hết $1$
          $\Rightarrow$ loại ngay bởi GCD test, không cần FM.
    \item \textbf{2 vòng lồng, mảng 2 chiều}: ghi $A[i_1][i_2]$, đọc
          $A[i_1-1][i_2+1]$ - hệ 4 biến, khả thi tại
          $\mathbf{i}_w=(0,1), \mathbf{i}_r=(1,0)$ $\Rightarrow$ có phụ thuộc.
\end{itemize}
\end{example}

\subsection{Vì sao tính trên trường số hữu tỉ \texorpdfstring{$\Q$}{Q}}

Lý thuyết Fourier--Motzkin giả định số học chính xác: mỗi bước khử chia cho hệ
số, dễ sinh phân số, và sai số dấu phẩy động sẽ tích lũy qua nhiều bước chiếu.
Phiên bản này dùng kiểu \texttt{fractions.Fraction} của Python, biểu diễn mọi
giá trị dưới dạng $p/q$ tối giản. Nhờ đó kết quả \emph{chính xác tuyệt đối}:
phân số như $1/3$ không bao giờ bị làm tròn, và hệ số ký hiệu hữu tỉ
($\sin(\pi/6) = 1/2$) được giữ đúng.

\subsection{Cấu trúc dữ liệu cốt lõi}

\begin{lstlisting}[caption={Lớp Row - một bất đẳng thức kèm chứng chỉ Farkas (fmlp/model.py)}]
@dataclass
class Row:

\subsection{Cách đánh giá đúng/sai}

Vì một bài toán LP có thể có \textbf{nhiều nghiệm tối ưu}, việc so trùng đúng
một bộ $(x_1, \ldots, x_n)$ là không hợp lý. Bộ test dùng hai tiêu chí độc lập:
\begin{enumerate}
    \item nghiệm trả về phải thỏa \emph{mọi} ràng buộc gốc (kiểm tra khả thi);
    \item giá trị $z^\ast$ phải khớp đáp án giải tay (so sánh \emph{chính xác}
          trên \texttt{Fraction}).
\end{enumerate}
Với bài toán đặc biệt, kiểm \emph{trạng thái} (\texttt{feasible} /
\texttt{infeasible} / \texttt{unbounded}) đúng kỳ vọng. Phương pháp hình học chỉ
áp dụng cho $n = 2$ và miền miền khả thi bị chặn; các case $n \ne 2$ hoặc không bị chặn
được đánh dấu \texttt{SKIP}. Parser kiểm bằng cách so từng hệ số \emph{chính
xác} (ví dụ $\sin(\pi/6)$ phải đúng $1/2$); riêng TXT1 kỳ vọng parser
\emph{báo lỗi} vì $\sqrt{2}$ không hữu tỉ.

TC10 & $\min \tfrac12 x_1{+}x_2$;\ $x_1{+}x_2\ge4$,\ $x_1\le6$,\ $x_i\ge0$ & $z^\ast = 2$ & PASS \\
\bottomrule
\end{tabularx}
\end{table}

\begin{infobox}[Nhận xét về nghiệm bội]
Ở TC2, phương pháp đại số trả về một đỉnh tối ưu hợp lệ cùng cho $z^\ast = 12$,
có thể khác đỉnh $x_1{=}x_2{=}x_3{=}2$ nhưng vẫn đúng - minh họa hiện tượng
nhiều nghiệm tối ưu. Tiêu chí đánh giá (khả thi $+$ đúng $z^\ast$) phản ánh
đúng bản chất này.
\end{infobox}

\subsection{So khớp hai phương pháp}

Với các bài toán hai biến miền khả thi bị chặn (TC1, TC3, TC5, TC6, TC10), phương pháp đại
số và hình học cho cùng $z^\ast$, xác nhận tính nhất quán. Phương pháp hình học
còn cung cấp toàn bộ đỉnh và dữ liệu vẽ để kiểm tra trực quan. Riêng TC9 (không
bị chặn) chỉ phương pháp đại số xử lý đúng - đúng như giới hạn đã nêu.

\subsection{Độ chính xác nhờ số học hữu tỉ}

Nhờ \texttt{Fraction}, các hệ số phân số được giữ nguyên dạng: TC5 cho
$z^\ast = 3$ chính xác; TC10 cho $z^\ast = 2$. Parser quy $\sin(\pi/6)$ và
$\cos(\pi/3)$ về đúng $1/2$ (TXT3), và \emph{từ chối} hệ số $\sqrt{2}$ vì không
hữu tỉ (TXT1) - một quyết định thiết kế nhất quán với việc làm việc trên
trường $\Q$.

\subsection{Phân tích độ phức tạp và thực nghiệm bùng nổ}
\label{sec:growth}

Như đã nêu ở \eqref{eq:blowup}, số ràng buộc có thể tăng theo hàm mũ kép
$O(m^{2^n})$. Để định lượng, ta đo số ràng buộc sau mỗi bước khử trên các hệ
\emph{dày} (hệ số dấu xen kẽ kèm hộp $0 \le x_j \le 6$), có và không có bước
lọc dư thừa ở Mục \ref{sec:prune}. Số liệu thực tế (script
\texttt{tests/growth.py}) ở Bảng \ref{tab:growth}.

thể giảm dưới mức mũ. Vì vậy Fourier--Motzkin phù hợp cho $n$ nhỏ và mang giá
trị \emph{minh họa, giải trình} hơn là giải LP quy mô lớn; Simplex hay điểm

\subsection{So sánh với việc hỏi trực tiếp một mô hình ngôn ngữ (LLM)}
\label{sec:vs-llm}

Một câu hỏi tự nhiên: tại sao không hỏi thẳng một công cụ như ChatGPT/GPT để
giải LP? Khác biệt cốt lõi nằm ở \emph{bản chất tính toán}, không chỉ ở tốc độ.

\begin{itemize}
    \item \textbf{Tất định và chính xác tuyệt đối.} Hệ thống chạy thuật toán
          Fourier--Motzkin trên trường $\Q$: mọi phép tính là phân số tối giản
          nên kết quả \emph{đúng tuyệt đối} và \emph{lặp lại được} --- cùng một
          input luôn cho cùng một output kèm nguyên vẹn chuỗi suy diễn. Ngược
          lại, LLM sinh văn bản theo xác suất nên có thể \emph{ảo giác}
          (hallucinate) ra bước biến đổi sai hoặc đáp số sai, và hai lần hỏi cùng
          một bài có thể trả về hai lời giải khác nhau.
    \item \textbf{Có thể kiểm chứng.} Mỗi nghiệm đều được thế ngược rồi kiểm lại
          trên toàn bộ ràng buộc gốc (Mục \ref{sec:growth} và tiêu chí ở Mục
          đánh giá), còn trường hợp vô nghiệm luôn kèm chứng chỉ Farkas. Lời giải
          do LLM sinh ra không đi kèm bất kỳ bảo chứng hình thức nào.
    \item \textbf{Phản hồi tức thì.} Với quy mô $n$ nhỏ mà bài tập nhắm tới, lời
          giải hiện ra \emph{ngay} khi bấm \textbf{Giải}: toàn bộ tính toán chạy
          cục bộ, không gọi mạng và không phải chờ mô hình sinh dần từng token
          như giao diện trò chuyện của LLM. Đây là lợi thế về trải nghiệm, tuy là
          thứ yếu so với hai điểm trên.
\end{itemize}

Để công bằng cũng cần nêu rõ giới hạn: lợi thế ``tức thì'' chỉ đúng với $n$ nhỏ
--- Fourier--Motzkin có thể bùng nổ mũ kép (Mục \ref{sec:growth}), nên với LP
quy mô lớn thì Simplex hay điểm trong vẫn vượt trội. Giá trị cốt lõi của hệ
thống là \emph{tính đúng đắn có giải trình}, chứ không phải tốc độ ở quy mô lớn.

\subsection{Cài đặt và chạy}

\begin{lstlisting}[language=bash, caption={Chạy ứng dụng ở local (Windows)}]
python -m venv .venv
.venv\Scripts\activate
pip install -r requirements.txt
python app.py
# Truy cap http://localhost:5001
\end{lstlisting}

\subsection{Cách nhập input}

\begin{enumerate}
    \item Chọn \textbf{số biến} $n$ và \textbf{kiểu mục tiêu} (MAX/MIN).
    \item Nhập \textbf{hệ số hàm mục tiêu} $c_1, \ldots, c_n$; ô xem trước hiển
          thị $z = \ldots$.
    \item Nhập \textbf{ràng buộc} theo một trong hai chế độ:
    \begin{itemize}
        \item \textbf{Form}: mỗi dòng gồm các ô hệ số, dấu ($\le, \ge, =$) và vế
              phải. Tùy chọn tự thêm điều kiện không âm $x_i \ge 0$.
        \item \textbf{Text}: gõ mỗi ràng buộc một dòng, ví dụ
              \texttt{2x1 + 3x2 <= 12}. Cho phép phân số và hằng lượng giác hữu
              tỉ trong hệ số (\texttt{sin}, \texttt{cos}, \texttt{tan},
              \texttt{cot}, \texttt{pi}). Không cho phép \texttt{x1\^{}2},
              \texttt{x1*x2}, \texttt{sin(x1)}, hay hệ số vô tỉ như
              \texttt{sqrt(2)}.
    \end{itemize}
    \item Chọn \textbf{phương pháp}: Đại số (luôn dùng được) hoặc Hình học (chỉ
          $n = 2$). Bấm \textbf{Giải}.
\end{enumerate}

\subsection{Các output trả về cho người dùng}

\begin{itemize}
    \item \textbf{ kết quả}: nghiệm $x_i^\ast$ (dạng phân số kèm thập phân),
          giá trị $z^\ast$, hoặc thông báo vô nghiệm / không bị chặn.
    \item \textbf{Tab ``Các bước khử''}: hệ ràng buộc ban đầu (sau khi gắn $z$)
          và sau mỗi bước khử biến.
    \item \textbf{Tab ``Biểu đồ''} (chỉ $n = 2$): miền khả thi, đường biên, các
          đỉnh, đường mức tối ưu, và bảng giá trị $z$ tại từng đỉnh.
    \item \textbf{Tab ``Diễn giải''}: từng bước suy luận chi tiết, gồm cả chứng chỉ Farkas khi vô nghiệm. Các bước khử được giải thích 4 bước nhỏ: cô lập biến, điều kiện tồn tại, ghép cặp cụ thể, giữ lại hệ rút gọn.
    \item \textbf{Tab ``Xuất file''}: xem trước và tải file \texttt{.txt} chứa
          toàn bộ giải trình (qua \texttt{POST /api/export}).
    \item \textbf{Khối ``Kiểm tra phụ thuộc vòng lặp''}: nhập số vòng lặp lồng
          $d$, số chiều mảng $m$, cận vòng lặp $[L_k, U_k]$, hàm chỉ số ghi và
          đọc (hệ số affine). Kết quả gồm: kết luận có/không phụ thuộc, nhân
          chứng cụ thể nếu có, giải thích ý nghĩa với vòng lặp đang nhập, và
          chi tiết các bước khử FM (ẩn, bấm mới hiện).
\end{itemize}

\subsection{API}

\subsection{So sánh với chatbot AI}

Hai bài toán được đưa cho ChatGPT với prompt yêu cầu giải bằng Fourier-Motzkin,
lọc dư thừa và xuất chứng chỉ Farkas nếu vô nghiệm. Kết quả được đối chiếu với
hệ thống. 
\newline Link chat với ChatGPT: https://chatgpt.com/share/6a1e0e7f-9670-83ec-bf5d-c65bfb578e20.
\newline 

\textbf{Bài 1 - TC5 (hệ số phân số):}
$\max z = x_1+x_2$, ràng buộc $\tfrac{1}{3}x_1+\tfrac{2}{3}x_2 \le 1$,
$x_1,x_2 \ge 0$. Cả hai đều cho $z^\ast = 3$.

\textbf{Bài 2 - Vô nghiệm với chứng chỉ Farkas phức tạp:}
$\max z = x_1+x_2$, ràng buộc $2x_1+x_2 \le 4$, $x_1+2x_2 \le 4$,
$x_1+x_2 \ge 5$, $x_1,x_2 \ge 0$. Chứng chỉ Farkas đòi hỏi tổ hợp
$1\cdot\text{rb}_1 + 1\cdot\text{rb}_2 + 3\cdot\text{rb}_3$ (hệ số không đều).
ChatGPT tìm đúng $\vy = (1,1,3,0,0)$ và kiểm tra $\vy\T\mA = \mathbf{0}$,
$\vy\T\vb = -7 < 0$ đúng chuẩn.

\textbf{Nhận xét tổng hợp:} với cả hai bài, ChatGPT cho kết quả đúng về mặt
toán học, kể cả chứng chỉ Farkas với hệ số phức tạp. Sự khác biệt giữa hệ
thống và ChatGPT không phải ở tính đúng đắn của thuật toán mà ở các yếu tố
sau:

\begin{itemize}
    \item \textbf{Tính lặp lại và xác định}: hệ thống luôn cho cùng kết quả với
          cùng đầu vào. ChatGPT có thể chọn thứ tự khử, cách trình bày khác
          nhau giữa các lần hỏi.
    \item \textbf{Kiểm tra nghiệm tự động}: hệ thống thế nghiệm vào toàn bộ
          ràng buộc gốc để xác nhận sau mỗi lần giải. ChatGPT có thể bỏ qua
          bước này.
    \item \textbf{Trực quan hóa}: hệ thống vẽ miền khả thi, đường mức trượt
          và đỉnh tối ưu nhấp nháy cho bài toán 2 biến.
    \item \textbf{Số học chính xác}: hệ thống làm việc trên trường $\Q$
          (\texttt{Fraction}), không bao giờ làm tròn. 
\end{itemize}

Ngược lại, ChatGPT linh hoạt hơn với bài toán phát biểu bằng ngôn ngữ tự nhiên
và giải thích khái niệm rộng hơn.

\subsection{Chức năng mở rộng đã làm}

\begin{itemize}
    \item Ứng dụng đã được triển khai trực tuyến bằng Render.com tại đường dẫn (có thể sẽ cần chờ một vài phút khi khởi động web): https://fourier-motzkin.onrender.com/
    \item Hai phương pháp giải song song (đại số + hình học) để đối chiếu.
    \item Số học hữu tỉ chính xác trên trường $\Q$; parser hệ số ký hiệu hữu tỉ.
    \item Chứng chỉ Farkas cho trường hợp vô nghiệm.
    \item Lọc dư thừa giảm bùng nổ tổ hợp, có thực nghiệm định lượng.
    \item Ứng dụng kiểm tra phụ thuộc vòng lặp tổng quát ($d$ vòng lồng, mảng
          $m$ chiều) với GCD test + FM, kèm giải thích ý nghĩa cụ thể.
    \item Biểu đồ hình học đầy đủ: tô miền nghiệm, mũi tên hướng miền, đường
          mức tối ưu nét liền, đường mức trượt animation, đỉnh tối ưu nhấp nháy.
    \item Bộ kiểm thử tự động (29 test) và xuất file giải trình.
\end{itemize}

\subsection{Hướng phát triển}

\begin{itemize}
    \item Loại dư thừa triệt để hơn (giải LP phụ) trước mỗi bước khử.
    \item Cho phương pháp hình học phát hiện trường hợp không bị chặn.
    \item Mở rộng kiểm tra phụ thuộc sang nghiệm nguyên chính xác bằng Omega
          test (Pugh, 1991) - khắc phục trường hợp nới lỏng thực báo thừa.
    \item Bổ sung Simplex để so sánh trực tiếp; trực quan hóa 3D cho $n = 3$.
\end{itemize}
